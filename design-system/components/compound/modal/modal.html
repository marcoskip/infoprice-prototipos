Modal

Janela modal sobre backdrop escurecido. 3 tamanhos (small 400px, medium 600px, large 800px). Estrutura: header (título + close), body (conteúdo), footer (ações).

Demo interativa 
<!-- Markup do modal (fica no body, fora do trigger) -->
<div class="modal-backdrop" id="myModal">
  <div class="modal">
    <div class="modal__header">
      <span class="modal__title">Confirmar ação</span>
      <button class="modal__close" data-modal-close>
        <span class="material-icons-outlined">close</span>
      </button>
    </div>
    <div class="modal__body">Conteudo...</div>
    <div class="modal__footer">
      <button class="btn btn--secondary-ghost" data-modal-close>Cancelar</button>
      <button class="btn btn--primary">Confirmar</button>
    </div>
  </div>
</div>

<!-- JS para abrir/fechar -->
<script>
const modal = document.getElementById('myModal');
document.getElementById('triggerBtn').onclick = () => modal.classList.add('is-open');
modal.querySelectorAll('[data-modal-close]').forEach(btn =>
  btn.onclick = () => modal.classList.remove('is-open')
);
modal.addEventListener('click', e => {
  if (e.target === modal) modal.classList.remove('is-open');
});
</script>

<!-- Tamanhos: .modal--small (400px) | (default 600px) | .modal--large (800px) -->
